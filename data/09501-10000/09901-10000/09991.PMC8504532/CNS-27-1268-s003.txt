#Import packages
library(meta)
library(dplyr)
library(dmetar)

#Load data
Data <- readxl::read_excel("Data.xlsx")
Data <- Data[order(Data$`Study ID`) , ]
GeneralData <- Data[order(Data$`Study ID`) , c("n" , "r")]

#Main synthesis
MainSynthesis <- metacor(GeneralData$r, GeneralData$n, studlab = Data$`Study ID`, comb.fixed = FALSE, text.w.random="Weight", method.tau = 'HS')
print(MainSynthesis)

#Create forest plot
png(file = "./Main Synthesis.png", width = 2800, height = 2100, res = 300)
forest.meta(MainSynthesis, xlim =  c(-1 , 1), test.overall = TRUE, calcwidth.tests = TRUE, leftlabs = c("Study ID" , "Sample Size"), test.overall.random = TRUE, prediction = TRUE, col.square = "red", col.diamond = "black")
while (!is.null(dev.list()))
dev.off()


#SMF subgroup analysis#
DataSMF <- subset(Data, !is.na(SMF))
GeneralDataSMF <- DataSMF[order(DataSMF$`Study ID`), c("n" , "r", "SMF")]
DataSMF <- DataSMF[order(DataSMF$`Study ID`), ]

SubGroupSMF <- metacor(GeneralDataSMF$r, GeneralDataSMF$n, studlab = DataSMF$`Study ID`, byvar = DataSMF$SMF, comb.fixed = FALSE, text.w.random="Weight", method.tau = 'HS')
print(SubGroupSMF)

png(file = "./SMF Subgroup.png", width = 3200, height = 2500, res = 300)
forest.meta(SubGroupSMF, xlim =  c(-1 , 1), leftlabs = c("Study ID" , "Sample Size"), print.byvar=FALSE, overall.hetstat=FALSE, test.subgroup.random = TRUE, calcwidth.tests = TRUE, col.square = "red", col.diamond = "black")
while (!is.null(dev.list()))
dev.off()
